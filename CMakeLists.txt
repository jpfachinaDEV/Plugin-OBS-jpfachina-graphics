cmake_minimum_required(VERSION 3.28)
project(jp-fachina-graphics VERSION 1.0.3 LANGUAGES CXX)
set(JP_RELEASE_VERSION "1.0.3-beta")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
option(JP_BUILD_PLUGIN "Build the OBS module" ON)
include(CTest)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Network WebSockets)
add_library(jp-core STATIC src/store.cpp src/store-operations.cpp src/project-settings.cpp src/composition-state.cpp src/scoreboard.cpp src/store.hpp src/server.cpp src/server.hpp src/downloader.cpp src/downloader.hpp resources.qrc)
target_link_libraries(jp-core PUBLIC Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network)
target_include_directories(jp-core PUBLIC src)
set_target_properties(jp-core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(jp-core PUBLIC /utf-8 /W4 /permissive-)
endif()
add_library(jp-ui STATIC src/controller.cpp src/controller.hpp src/obs-websocket.cpp src/obs-websocket.hpp src/operator-window.cpp src/operator-window.hpp src/operator-panel.cpp src/scoreboard-workspace.cpp src/operator-panel.hpp src/operator-dialogs.cpp src/scoreboard-dialog.cpp src/project-dialogs.cpp src/category-sync-dialog.cpp src/position-canvas.cpp src/position-editor.cpp src/position-editor.hpp src/media-card.cpp src/media-card.hpp)
target_link_libraries(jp-ui PUBLIC jp-core Qt6::WebSockets)
if(WIN32)
  target_link_libraries(jp-ui PUBLIC Crypt32)
endif()
if(JP_BUILD_PLUGIN)
  if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "The OBS plugin requires Windows x64 and MSVC (matching OBS Qt ABI).")
  endif()
  find_package(libobs 32.1 REQUIRED CONFIG)
  find_package(obs-frontend-api REQUIRED CONFIG)
  if(libobs_VERSION VERSION_GREATER_EQUAL 33 AND Qt6_VERSION VERSION_LESS 6.11.1)
    message(FATAL_ERROR "OBS 33 requires the matching Qt 6.11.1 kit, including Qt WebSockets. Do not package the Qt 6.8 module with OBS 33.")
  endif()
  file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/jp-build-info.json" CONTENT "{\n  \"pluginVersion\": \"${JP_RELEASE_VERSION}\",\n  \"binaryVersion\": \"${PROJECT_VERSION}\",\n  \"releaseChannel\": \"beta\",\n  \"obsVersion\": \"${libobs_VERSION}\",\n  \"qtVersion\": \"${Qt6_VERSION}\",\n  \"architecture\": \"x64\"\n}\n")
  add_library(jp-fachina-graphics MODULE src/plugin-main.cpp src/obs-native.cpp)
  target_link_libraries(jp-fachina-graphics PRIVATE jp-ui OBS::libobs OBS::obs-frontend-api)
  if(BUILD_TESTING)
    add_executable(jp-render-tests tests/render-test.cpp src/obs-native.cpp)
    target_link_libraries(jp-render-tests PRIVATE jp-core OBS::libobs Psapi)
  endif()
  install(TARGETS jp-fachina-graphics LIBRARY DESTINATION obs-plugins/64bit RUNTIME DESTINATION obs-plugins/64bit)
  get_filename_component(_qtBase "${Qt6_DIR}/../../.." ABSOLUTE)
  install(FILES "${_qtBase}/bin/Qt6WebSockets.dll" DESTINATION obs-plugins/64bit)
  set(JP_QT_PLUGIN_DIR "${_qtBase}/plugins" CACHE PATH "Matching Qt plugin directory")
  install(FILES "${JP_QT_PLUGIN_DIR}/tls/qschannelbackend.dll" DESTINATION data/obs-plugins/jp-fachina-graphics/qt-plugins/tls)
  install(FILES vendor/Qt-LGPL-3.0.txt vendor/Qt-GPL-3.0.txt vendor/Qt-NOTICE.md vendor/obs-browser/COPYING DESTINATION data/obs-plugins/jp-fachina-graphics/licenses)
endif()
add_executable(jp-graphics-dev src/dev-main.cpp)
target_link_libraries(jp-graphics-dev PRIVATE jp-ui)
if(BUILD_TESTING)
  add_executable(jp-core-tests tests/core-test.cpp)
  target_link_libraries(jp-core-tests PRIVATE jp-core)
  add_test(NAME core COMMAND jp-core-tests)
  set_tests_properties(core PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=minimal")
  add_executable(jp-operator-tests tests/operator-test.cpp)
  target_link_libraries(jp-operator-tests PRIVATE jp-ui)
  add_test(NAME operator COMMAND jp-operator-tests)
  set_tests_properties(operator PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=minimal" WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
  add_executable(jp-panel-tests tests/panel-test.cpp)
  target_link_libraries(jp-panel-tests PRIVATE jp-ui)
  add_test(NAME panel COMMAND jp-panel-tests)
  set_tests_properties(panel PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=minimal" WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
  add_executable(jp-position-tests tests/position-project-test.cpp)
  target_link_libraries(jp-position-tests PRIVATE jp-ui)
  add_test(NAME position COMMAND jp-position-tests)
  set_tests_properties(position PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=minimal" WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
  add_executable(jp-websocket-tests tests/websocket-test.cpp)
  target_link_libraries(jp-websocket-tests PRIVATE jp-ui)
  add_test(NAME websocket COMMAND jp-websocket-tests)
  set_tests_properties(websocket PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=minimal" WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
endif()
